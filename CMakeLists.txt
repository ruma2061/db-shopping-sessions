cmake_minimum_required(VERSION 3.20)
project(shopping_sessions LANGUAGES CXX)
add_executable(sessions-core src/main.cpp)
target_compile_features(sessions-core PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(sessions-core PRIVATE /W4)
else()
  target_compile_options(sessions-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
